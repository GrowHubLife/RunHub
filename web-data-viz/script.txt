create database ProjetoPessoal;

use ProjetoPessoalra;

create table usuario (
id int primary key auto_increment,
nome varchar(45),
email varchar(45),
senha varchar(45)
);


CREATE TABLE comunidade (
	id INT PRIMARY KEY AUTO_INCREMENT,
	titulo VARCHAR(100),
	descricao VARCHAR(150),
	fk_usuario INT,
	FOREIGN KEY (fk_usuario) REFERENCES usuario(id)
);

CREATE TABLE registro (
    id INT PRIMARY KEY AUTO_INCREMENT,
    data_treino DATE,
    distancia_km INT,
    tipo_treino VARCHAR(20),
    CONSTRAINT chkTipoTreino CHECK (tipo_treino IN ('Leve', 'Intervalado', 'Tiro', 'Longão', 'Caminhada')),
    batimentos_medios INT,
	pace INT,
    calorias INT,
    sensacao_do_dia INT,
    CONSTRAINT chkSensacaoDoDia CHECK (sensacao_do_dia BETWEEN 1 AND 10),
    observacoes TEXT,
	fkUsuario int,
    foreign key (fkUsuario)
    references usuario(id)
);
